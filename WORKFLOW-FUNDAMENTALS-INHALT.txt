name: Fundamentaldaten aktualisieren

on:
  schedule:
    - cron: "30 5 * * *"      # taeglich 05:30 UTC
  workflow_dispatch:          # erlaubt manuelles Ausloesen ueber die Actions-Seite

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Python einrichten
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Fundamentaldaten holen
        env:
          ALPHAVANTAGE_KEY: ${{ secrets.ALPHAVANTAGE_KEY }}
        run: python3 scripts/build_fundamentals.py

      - name: Aenderungen committen
        run: |
          git config user.name "github-actions"
          git config user.email "actions@github.com"
          git add data/fundamentals.json
          git diff --staged --quiet || git commit -m "Fundamentaldaten aktualisiert"
          git push
