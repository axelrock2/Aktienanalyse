name: Chancenraum aktualisieren

on:
  schedule:
    - cron: "45 5 * * 1-5"    # werktags 05:45 UTC, nach Boersenschluss USA
  workflow_dispatch: {}

permissions:
  contents: write

concurrency:
  group: chancen
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Python einrichten
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Chancenraum rechnen
        run: python3 scripts/build_chancen.py

      - name: Aenderungen committen
        run: |
          git config user.name "chancen-bot"
          git config user.email "actions@users.noreply.github.com"
          git add data/chancen.json
          git diff --cached --quiet && echo "Keine Aenderungen." && exit 0
          git commit -m "Chancenraum aktualisiert ($(date -u +%F))"
          git pull --rebase --autostash || true
          git push
